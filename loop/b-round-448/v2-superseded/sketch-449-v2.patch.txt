+function listFiles(root: string, extensions: RegExp): string[] {
+  const files: string[] = [];
+  const walk = (directory: string) => {
+    for (const entry of readdirSync(directory)) {
+      const path = join(directory, entry);
+      if (statSync(path).isDirectory()) {
+        if (entry !== '__tests__') {
+          walk(path);
+        }
+      } else if (extensions.test(entry)) {
+        files.push(relative(root, path).split('\\').join('/'));

+function storeCss(path: string): string {

+function stripComments(css: string): string {

+function gradientTextHits(css: string): CssDeclarationHit[] {
+  const source = stripComments(css);
+  const hits: CssDeclarationHit[] = [];
+  const selectors: string[] = [];
+  let segmentStart = 0;
+  for (let index = 0; index < source.length; index += 1) {
+    const character = source[index];
+    if (character === '{') {
+      selectors.push(source.slice(segmentStart, index).trim());
+      segmentStart = index + 1;
+    } else if (character === '}' || character === ';') {
+      const segment = source.slice(segmentStart, index);
+      for (const match of segment.matchAll(GRADIENT_TEXT_PATTERN)) {
+        hits.push({ selector: selectors[selectors.length - 1] ?? '', declaration: match[0] });
+      }
+      if (character === '}') {
+        selectors.pop();
+      }
+      segmentStart = index + 1;
+    }
+  }
+  return hits;
+}
