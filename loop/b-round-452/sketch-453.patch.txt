+function ruleBodiesFor(css: string, selector: string): string[] {
+  const source = css.replace(/\/\*[\s\S]*?\*\//g, ' ');
+  const bodies: string[] = [];
+  for (const match of source.matchAll(/([^{}]*)\{([^{}]*)\}/g)) {
+    const targets = (match[1] ?? '').split(',').map(
+      (part) =>
+        part
+          .trim()

+          .pop() ?? '',

+    if (targets.includes(selector)) {
+      bodies.push(match[2] ?? '');
+    }
+  }
