+function countMatches(pattern: RegExp): (source: string) => number {

+function topLevelRuleBodies(css: string, selector: string): string[] {
+  const bodies: string[] = [];
+  let lineStart = 0;
+  for (const line of css.split('\n')) {
+    if (line === `${selector} {`) {
+      const bodyStart = lineStart + line.length;
+      const bodyEnd = css.indexOf('\n}', bodyStart);
+      bodies.push(css.slice(bodyStart, bodyEnd));
+    }
+    lineStart += line.length + 1;
+  }
+  return bodies;
+}
