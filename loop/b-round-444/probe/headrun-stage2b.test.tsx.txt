function imeArrangement(): Arrangement {
  const base = createArrangement([
    { id: 'video', kind: 'video', name: 'Video' },
    { id: 'music', kind: 'audio', name: 'Music' },
    { id: 'sfx', kind: 'audio', name: 'SFX' },
  ]);
  return {
    ...base,
    tracks: base.tracks.map((track) =>
      track.id === 'music' || track.id === 'sfx' ? { ...track, groupId: 'stems' } : track,
    ),
    groups: [{ id: 'stems', name: 'Stems', parentId: null }],
    clips: [{ id: 'h1', trackId: 'sfx', sourceId: 'hit', start: 0, sourceOffset: 0, duration: 16 }],
  };
}

interface ImeApiRef {
  current: WorkspaceProbeValue | null;
}

function ImeProbe({ api }: { api: ImeApiRef }) {
  api.current = useWorkspaceProbe();
  return null;
}

function live(api: ImeApiRef): WorkspaceProbeValue {
  if (!api.current) {
    throw new Error('provider not mounted');
  }
  return api.current;
}

async function mountImeWorkspace() {
  const api: ImeApiRef = { current: null };
  const view = render(
    <ArrangementProvider collection={imeCollection} initialArrangement={imeArrangement()}>
      <StudioWorkspace />
      <ImeProbe api={api} />
    </ArrangementProvider>,
  );
  await act(async () => {
    await Promise.resolve();
    await Promise.resolve();
  });
  const snapshot = () => ({
    arrangement: JSON.stringify(live(api).getArrangement()),
    undo: live(api).undoLabel,
    status: live(api).status,
    statusSequence: live(api).statusSequence,
  });
  return { api, view, snapshot };
}

type EditorKey = 'Enter' | 'Escape';
type EditorId = 'tempo' | 'marker' | 'group' | 'range' | 'gain';
type KeyForm = 'composing' | 'held' | 'plain';
type ApiValue = Awaited<ReturnType<typeof mountImeWorkspace>>;

interface KeyCell {
  prevented: boolean;
  documentSaw: boolean;
}

const EDITORS: readonly EditorId[] = ['tempo', 'marker', 'group', 'range', 'gain'];

function inputById(view: ApiValue['view'], testId: string): HTMLInputElement {
  const element = view.getByTestId(testId);
  if (!(element instanceof HTMLInputElement)) {
    throw new Error(`${testId} is not an input`);
  }
  return element;
}

function openEditor(mounted: ApiValue, editor: EditorId): HTMLInputElement {
  const { api, view } = mounted;
  if (editor === 'tempo') {
    const tempo = inputById(view, 'arrangement-tempo-input');
    act(() => tempo.focus());
    fireEvent.change(tempo, { target: { value: '96' } });
    return tempo;
  }
  if (editor === 'marker') {
    fireEvent.click(view.getByTestId('arrangement-marker-add'));
    const marker = inputById(view, 'arrangement-marker-name-input');
    fireEvent.change(marker, { target: { value: 'Hit' } });
    return marker;
  }
  if (editor === 'group') {
    const name = view.getByTestId('arrangement-group-name');
    act(() => name.focus());
    fireEvent.keyDown(name, { key: 'F2', code: 'F2' });
    const group = inputById(view, 'arrangement-group-name-input');
    fireEvent.change(group, { target: { value: 'Bed' } });
    return group;
  }
  act(() => live(api).selectClip('h1'));
  const field = inputById(
    view,
    editor === 'range' ? 'arrangement-detail-start' : 'arrangement-detail-clip-gain-input',
  );
  act(() => field.focus());
  fireEvent.change(field, { target: { value: editor === 'range' ? '2' : '-3' } });
  return field;
}

function pressCell(target: Element, key: EditorKey, form: KeyForm): KeyCell {
  let documentSaw = false;
  const listener = () => {
    documentSaw = true;
  };
  document.addEventListener('keydown', listener);
  const init =
    form === 'composing'
      ? { key, code: key, isComposing: true }
      : form === 'held'
        ? { key, code: key, repeat: true }
        : { key, code: key };
  const notPrevented = fireEvent.keyDown(target, init);
  document.removeEventListener('keydown', listener);
  return { prevented: !notPrevented, documentSaw };
}

const CELL_ORDER: readonly (readonly [EditorKey, KeyForm])[] = [
  ['Enter', 'composing'],
  ['Escape', 'composing'],
  ['Enter', 'held'],
  ['Escape', 'held'],
  ['Escape', 'plain'],
  ['Enter', 'plain'],
];

const SEEN_PREVENTED: KeyCell = { prevented: true, documentSaw: true };
const SEEN_OPEN: KeyCell = { prevented: false, documentSaw: true };
const STOPPED_PREVENTED: KeyCell = { prevented: true, documentSaw: false };
const STOPPED_OPEN: KeyCell = { prevented: false, documentSaw: false };

const HEAD_TRUTH_TABLE: Readonly<Record<EditorId, readonly KeyCell[]>> = {
  tempo: [SEEN_OPEN, SEEN_OPEN, SEEN_PREVENTED, SEEN_PREVENTED, SEEN_PREVENTED, SEEN_PREVENTED],
  marker: [
    STOPPED_OPEN,
    STOPPED_OPEN,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
  ],
  group: [
    STOPPED_OPEN,
    STOPPED_OPEN,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
    STOPPED_PREVENTED,
  ],
  range: [SEEN_OPEN, SEEN_OPEN, SEEN_PREVENTED, SEEN_OPEN, SEEN_OPEN, SEEN_PREVENTED],
  gain: [SEEN_OPEN, SEEN_OPEN, SEEN_PREVENTED, SEEN_PREVENTED, SEEN_PREVENTED, SEEN_PREVENTED],
};

function holdPlain(target: Element, key: EditorKey) {
  for (let step = 0; step < 3; step += 1) {
    fireEvent.keyDown(target, { key, code: key, repeat: true });
  }
}

function gainDb(api: ApiValue['api']): number {
  return (
    live(api)
      .getArrangement()
      .clips.find((clip) => clip.id === 'h1')?.gainDb ?? 0
  );
}

describe('inline editors share one settle-key rule (SYS-3 stage 2b)', () => {
  it.each(
    EDITORS,
  )('IME12 %s editor: composing, held and plain Enter/Escape keep the HEAD 616729c truth table (defaultPrevented, reaches document)', async (editor) => {
    const mounted = await mountImeWorkspace();
    const seen: KeyCell[] = [];
    let target = openEditor(mounted, editor);
    for (const [key, form] of CELL_ORDER) {
      if (!target.isConnected || document.activeElement !== target) {
        target = openEditor(mounted, editor);
      }
      seen.push(pressCell(target, key, form));
    }
    expect(seen).toStrictEqual(HEAD_TRUTH_TABLE[editor]);
  });

  it('IME13 BPM field: plain held Enter or Escape (repeat only, no IME) keeps focus and the draft; a fresh Escape then drops the draft', async () => {
    const mounted = await mountImeWorkspace();
    const before = mounted.snapshot();
    const tempo = openEditor(mounted, 'tempo');
    holdPlain(tempo, 'Enter');
    holdPlain(tempo, 'Escape');
    expect(document.activeElement).toBe(tempo);
    expect(tempo.value).toBe('96');
    expect(mounted.snapshot()).toStrictEqual(before);
    expect(fireEvent.keyDown(tempo, { key: 'Escape', code: 'Escape' })).toBe(false);
    expect(document.activeElement).not.toBe(tempo);
    expect(tempo.value).toBe('');
    expect(mounted.snapshot()).toStrictEqual(before);
  });

  it('IME13 marker name editor: plain held Enter or Escape keeps the editor open and adds nothing; a fresh Enter then adds the marker once', async () => {
    const mounted = await mountImeWorkspace();
    const before = mounted.snapshot();
    const marker = openEditor(mounted, 'marker');
    holdPlain(marker, 'Enter');
    holdPlain(marker, 'Escape');
    expect(mounted.view.getByTestId('arrangement-marker-name-input')).toBe(marker);
    expect(document.activeElement).toBe(marker);
    expect(marker.value).toBe('Hit');
    expect(mounted.snapshot()).toStrictEqual(before);
    fireEvent.keyDown(marker, { key: 'Enter', code: 'Enter' });
    expect(mounted.view.queryByTestId('arrangement-marker-name-input')).toBeNull();
    expect(
      (live(mounted.api).getArrangement().markers ?? []).map((entry) => entry.name),
    ).toStrictEqual(['Hit']);
  });

  it('IME13 group name editor: plain held Enter or Escape keeps the editor open and renames nothing; a fresh Enter then renames once', async () => {
    const mounted = await mountImeWorkspace();
    const before = mounted.snapshot();
    const group = openEditor(mounted, 'group');
    holdPlain(group, 'Enter');
    holdPlain(group, 'Escape');
    expect(mounted.view.getByTestId('arrangement-group-name-input')).toBe(group);
    expect(document.activeElement).toBe(group);
    expect(group.value).toBe('Bed');
    expect(mounted.snapshot()).toStrictEqual(before);
    fireEvent.keyDown(group, { key: 'Enter', code: 'Enter' });
    expect(mounted.view.queryByTestId('arrangement-group-name-input')).toBeNull();
    expect(live(mounted.api).undoLabel).toBe('Rename Group');
    expect(
      (live(mounted.api).getArrangement().groups ?? []).map((entry) => entry.name),
    ).toStrictEqual(['Bed']);
  });

  it('IME13 Detail From field: plain held Enter applies no time selection; a fresh Enter then applies it', async () => {
    const mounted = await mountImeWorkspace();
    const start = openEditor(mounted, 'range');
    const before = mounted.snapshot();
    holdPlain(start, 'Enter');
    expect(document.activeElement).toBe(start);
    expect(start.value).toBe('2');
    expect(mounted.snapshot()).toStrictEqual(before);
    expect(fireEvent.keyDown(start, { key: 'Enter', code: 'Enter' })).toBe(false);
    expect(mounted.snapshot().statusSequence).toBeGreaterThan(before.statusSequence);
    expect(mounted.snapshot().status).toMatch(/^Time selection: 00:02\.000 — /);
  });

  it('IME13 Clip Gain field: plain held Enter or Escape applies and reverts nothing; a fresh Escape then shows the stored gain again', async () => {
    const mounted = await mountImeWorkspace();
    act(() => live(mounted.api).selectClip('h1'));
    const shown = inputById(mounted.view, 'arrangement-detail-clip-gain-input').value;
    const gain = openEditor(mounted, 'gain');
    const before = mounted.snapshot();
    holdPlain(gain, 'Enter');
    holdPlain(gain, 'Escape');
    expect(document.activeElement).toBe(gain);
    expect(gain.value).toBe('-3');
    expect(mounted.snapshot()).toStrictEqual(before);
    expect(fireEvent.keyDown(gain, { key: 'Escape', code: 'Escape' })).toBe(false);
    expect(gain.value).toBe(shown);
    expect(gainDb(mounted.api)).toBe(0);
  });
});
