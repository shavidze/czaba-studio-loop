+// @vitest-environment node
+import { describe, expect, it } from 'vitest';
+import { readApp } from './helpers/style-scan';
+
+const CART_PATH = 'routes/cart.tsx';
+const PAGE_HEIGHT = 'h-[calc(100vh-80px)]';
+const READER_HINT =
+  'cart.tsx keeps a plain className="..." here; update the reader if it moves to cn()';
+const BELOW_LG_HEIGHT_OR_CLIP =
+  /^(?:(?:sm|md):)?(?:h|max-h)-|^(?:(?:sm|md):)?overflow(?:-[xy])?-(?:hidden|clip)$/;
+
+function classTokensOfTags(source: string, tagName: string): string[][] {
+  const pattern = new RegExp(`<${tagName}\\s+className="([^"]*)"`, 'g');
+  return [...source.matchAll(pattern)].map((match) => match[1].split(/\s+/).filter(Boolean));
+}
+
+describe('cart page layout below lg (B-R502-CART-MOBILE-ITEMS-HIDDEN)', () => {
+  const source = readApp(CART_PATH);
+
+  it('CML1: below lg the page grows with its content; the fixed height and the clipping are lg-only', () => {
+    const mains = classTokensOfTags(source, 'main');
+    expect(mains, READER_HINT).toHaveLength(1);
+    const [tokens] = mains;
+    expect(tokens).toContain(`min-${PAGE_HEIGHT}`);
+    expect(tokens).toContain(`lg:${PAGE_HEIGHT}`);
+    expect(tokens).toContain('lg:overflow-hidden');
+    expect(tokens.filter((token) => BELOW_LG_HEIGHT_OR_CLIP.test(token))).toEqual([]);
+  });
+
+  it('CML2: at lg the grid row is bounded and each column scrolls on its own under the mouse wheel', () => {
+    const divs = classTokensOfTags(source, 'div');
+    const grids = divs.filter((tokens) => tokens.includes('lg:grid-cols-3'));
+    expect(grids, READER_HINT).toHaveLength(1);
+    expect(grids[0]).toEqual(
+      expect.arrayContaining(['grid', 'grid-cols-1', 'lg:grid-rows-[minmax(0,1fr)]', 'min-h-0']),
+    );
+    const items = divs.filter((tokens) => tokens.includes('lg:col-span-2'));
+    expect(items, READER_HINT).toHaveLength(1);
+    expect(items[0]).toEqual(expect.arrayContaining(['overflow-y-auto', 'overscroll-contain']));
+    const summary = divs.filter((tokens) => tokens.includes('lg:col-span-1'));
+    expect(summary, READER_HINT).toHaveLength(1);
+    expect(summary[0]).toEqual(
+      expect.arrayContaining(['lg:min-h-0', 'lg:overflow-y-auto', 'lg:overscroll-contain']),
+    );
+    expect(summary[0].filter((token) => BELOW_LG_HEIGHT_OR_CLIP.test(token))).toEqual([]);
+    const wheelColumns = [
+      ...source.matchAll(/<div\s+className="([^"]*)"\s+data-lenis-prevent-wheel\s*>/g),
+    ].map((match) => match[1].split(/\s+/)[0]);
+    expect(wheelColumns).toEqual(['lg:col-span-2', 'lg:col-span-1']);
+  });
+
+  it('CML4: the summary card is not sticky inside its scrolling column', () => {
+    const cards = classTokensOfTags(source, 'div').filter(
+      (tokens) => tokens.includes('bg-surface') && tokens.includes('p-8'),
+    );
+    expect(cards, READER_HINT).toHaveLength(1);
+    expect(cards[0].filter((token) => /(^|:)(sticky|top-\d+)$/.test(token))).toEqual([]);
+  });
+
+  it('CML3: the self-test reads a multi-token main and rejects the HEAD layout', () => {
+    const head = '<main className="h-[calc(100vh-80px)] bg-background overflow-hidden">';
+    const [headTokens] = classTokensOfTags(head, 'main');
+    expect(headTokens).toEqual([PAGE_HEIGHT, 'bg-background', 'overflow-hidden']);
+    expect(headTokens.filter((token) => BELOW_LG_HEIGHT_OR_CLIP.test(token))).toEqual([
+      PAGE_HEIGHT,
+      'overflow-hidden',
+    ]);
+    const allowed = ['min-h-[calc(100vh-80px)]', 'lg:h-[calc(100vh-80px)]', 'lg:overflow-hidden'];
+    expect(allowed.filter((token) => BELOW_LG_HEIGHT_OR_CLIP.test(token))).toEqual([]);
+    const banned = [
+      'md:h-[calc(100vh-80px)]',
+      'md:overflow-hidden',
+      'h-screen',
+      'max-h-screen',
+      'overflow-y-clip',
+    ];
+    expect(banned.filter((token) => BELOW_LG_HEIGHT_OR_CLIP.test(token))).toEqual(banned);
+  });
+});

+1. **ტელეფონზე სია არ ჩანს (High).** 1024px-ზე ვიწრო ეკრანზე სვეტები ერთმანეთის ქვეშ ლაგდება და პროდუქტების სვეტს მხოლოდ ის ადგილი რჩება, რასაც Order Summary არ იკავებს (`head-cart.json`).
