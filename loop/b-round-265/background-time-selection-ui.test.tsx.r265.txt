const PX_PER_SECOND = 40;
const LANE_HEIGHT_PX = 100;
const HEADER_WIDTH_PX = 224;
const MUSIC_LANE_Y = 150;
const STEMS_LANE_Y = 250;
const TRACK_IDS = ['video', 'music', 'stems', 'sfx'] as const;

const backgroundCollection: WorkspaceCollection = {
  id: 'background',
  label: 'background',
  origin: 'demo',
  sources: [
    { id: 'song', title: 'Song', collectionLabel: 'background', kind: 'music', origin: 'demo' },
  ],
  loadSource: async () => new AudioBuffer({ length: 4000, numberOfChannels: 1, sampleRate: 100 }),
};

function clipAt(id: string, trackId: string, start: number, duration: number): ArrangementClip {
  return { id, trackId, sourceId: 'song', start, sourceOffset: 0, duration };
}

const TWO_MUSIC_CLIPS = [clipAt('m1', 'music', 10, 10), clipAt('m2', 'music', 25, 5)];

function backgroundArrangement(
  clips: ArrangementClip[],
  markers: ArrangementMarker[],
): Arrangement {
  const base = createArrangement([
    { id: 'video', kind: 'video', name: 'Video' },
    { id: 'music', kind: 'audio', name: 'Music' },
    { id: 'stems', kind: 'audio', name: 'Stems' },
    { id: 'sfx', kind: 'audio', name: 'SFX' },
  ]);
  return { ...base, clips, markers };
}

interface BackgroundApiRef {
  current: ArrangementContextValue | null;
  position: ArrangementPositionStore | null;
}

function BackgroundProbe({ api }: { api: BackgroundApiRef }) {
  api.current = useArrangement();
  api.position = useArrangementPositionStore();
  return null;
}

function live(api: BackgroundApiRef): ArrangementContextValue {
  if (!api.current) {
    throw new Error('provider not mounted');
  }
  return api.current;
}

function positionOf(api: BackgroundApiRef): number {
  if (!api.position) {
    throw new Error('position store not mounted');
  }
  return api.position.getPosition();
}

function stubCapture(element: HTMLElement) {
  Object.defineProperty(element, 'setPointerCapture', { configurable: true, value: () => {} });
  Object.defineProperty(element, 'hasPointerCapture', { configurable: true, value: () => false });
  Object.defineProperty(element, 'releasePointerCapture', { configurable: true, value: () => {} });
}

async function mountBackgroundWorkspace(
  clips: ArrangementClip[],
  markers: ArrangementMarker[] = [],
) {
  const api: BackgroundApiRef = { current: null, position: null };
  const view = render(
    <ArrangementProvider
      collection={backgroundCollection}
      initialArrangement={backgroundArrangement(clips, markers)}
    >
      <StudioWorkspace />
      <BackgroundProbe api={api} />
    </ArrangementProvider>,
  );
  for (let tick = 0; tick < 10 && live(api).sourceDurations.get('song') === undefined; tick++) {
    await act(async () => {
      await Promise.resolve();
      await Promise.resolve();
    });
  }
  act(() => live(api).setPxPerSecond(PX_PER_SECOND));
  view.container.querySelectorAll<HTMLElement>('[data-track-id]').forEach((row, index) => {
    const top = index * LANE_HEIGHT_PX;
    Object.defineProperty(row, 'getBoundingClientRect', {
      configurable: true,
      value: () => new DOMRect(0, top, 800, LANE_HEIGHT_PX),
    });
  });
  const laneBody = (trackId: string) => {
    const body = view.container.querySelector<HTMLElement>(
      `[data-track-id="${trackId}"] [data-grid-seconds]`,
    );
    if (!body) {
      throw new Error(`lane ${trackId} not rendered`);
    }
    return body;
  };
  const clipParts = (clipId: string) => {
    const root = view.container.querySelector<HTMLElement>(`[data-clip-id="${clipId}"]`);
    const body = root?.querySelector<HTMLElement>('[title^="drag = time selection"]');
    if (!root || !body) {
      throw new Error(`clip ${clipId} not rendered`);
    }
    stubCapture(root);
    return { root, body };
  };
  const laneRanges = (trackId: string) =>
    Array.from(
      view.container.querySelectorAll<HTMLElement>(
        `[data-track-id="${trackId}"] [data-testid="arrangement-lane-range"]`,
      ),
    );
  return { api, view, laneBody, clipParts, laneRanges };
}

type BackgroundMount = Awaited<ReturnType<typeof mountBackgroundWorkspace>>;

const press = { button: 0, buttons: 1 };

function xFor(time: number): number {
  return Math.round(time * PX_PER_SECOND);
}

interface BackgroundDragOptions {
  trackId?: string;
  startY?: number;
  moveY?: number;
  ctrlKey?: boolean;
  metaKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  release?: boolean;
}

function dragBackground(
  mounted: BackgroundMount,
  fromX: number,
  toXs: number[],
  options: BackgroundDragOptions = {},
) {
  const startY = options.startY ?? MUSIC_LANE_Y;
  const moveY = options.moveY ?? startY;
  fireEvent.pointerDown(mounted.laneBody(options.trackId ?? 'music'), {
    ...press,
    pointerId: 1,
    clientX: fromX,
    clientY: startY,
    ctrlKey: options.ctrlKey ?? false,
    metaKey: options.metaKey ?? false,
    shiftKey: options.shiftKey ?? false,
  });
  for (const x of toXs) {
    fireEvent.pointerMove(window, {
      pointerId: 1,
      clientX: x,
      clientY: moveY,
      altKey: options.altKey ?? false,
    });
  }
  if (options.release === false) {
    return;
  }
  fireEvent.pointerUp(window, {
    pointerId: 1,
    clientX: toXs.at(-1) ?? fromX,
    clientY: moveY,
    altKey: options.altKey ?? false,
  });
}

function dragClipBody(mounted: BackgroundMount, clipId: string, fromX: number, toXs: number[]) {
  const { root, body } = mounted.clipParts(clipId);
  fireEvent.pointerDown(body, { ...press, pointerId: 1, clientX: fromX, clientY: MUSIC_LANE_Y });
  for (const x of toXs) {
    fireEvent.pointerMove(root, { pointerId: 1, clientX: x, clientY: MUSIC_LANE_Y });
  }
  fireEvent.pointerUp(root, {
    pointerId: 1,
    clientX: toXs.at(-1) ?? fromX,
    clientY: MUSIC_LANE_Y,
  });
}

function laneExtentSeconds(mounted: BackgroundMount): number {
  return Number.parseFloat(mounted.laneBody('music').style.width) / PX_PER_SECOND;
}

function expectOnlyMusicOverlay(mounted: BackgroundMount, start: string, end: string) {
  const overlays = mounted.laneRanges('music');
  expect(overlays).toHaveLength(1);
  expect(overlays[0]?.getAttribute('data-range-start')).toBe(start);
  expect(overlays[0]?.getAttribute('data-range-end')).toBe(end);
  expect(overlays[0]?.getAttribute('aria-hidden')).toBe('true');
  expect(overlays[0]?.textContent).not.toContain(studioEditStrings.allTracks);
  for (const trackId of TRACK_IDS.filter((id) => id !== 'music')) {
    expect(mounted.laneRanges(trackId), trackId).toHaveLength(0);
  }
}

function pressKey(init: KeyboardEventInit) {
  fireEvent.keyDown(document.body, init);
}

function statusText(mounted: BackgroundMount): string | null {
  return mounted.view.getByTestId('arrangement-status').textContent;
}

describe('SABA-B-11c stage 1 background time selection', () => {
  it('BT1 a plain background drag makes a free lane range, clears the selection and activates the lane', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const { api, view } = mounted;
    act(() => live(api).selectClip('m2'));
    act(() => live(api).setActiveTrack('sfx'));
    expect(live(api).selectedClipIds).toEqual(['m2']);
    dragBackground(mounted, xFor(1), [xFor(1) + 20, xFor(4), xFor(7.5)], { release: false });
    expect(view.queryByTestId('arrangement-marquee')).toBeNull();
    expect(live(api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    fireEvent.pointerUp(window, { pointerId: 1, clientX: xFor(7.5), clientY: MUSIC_LANE_Y });
    expect(view.queryByTestId('arrangement-marquee')).toBeNull();
    expect(live(api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '1.000', '7.500');
    expect(live(api).selectedClipIds).toEqual([]);
    expect(live(api).activeTrackId).toBe('music');
    expect(view.getByTestId('arrangement-insert-marker').hidden).toBe(true);
    expect(view.getByTestId('arrangement-range').textContent).toContain('00:01.000');
  });

  it('BT1 a Shift drag also makes a range, and a drag to the left gives the same ordered range', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragBackground(mounted, xFor(7.5), [xFor(1)], { shiftKey: true });
    expect(mounted.view.queryByTestId('arrangement-marquee')).toBeNull();
    expect(live(mounted.api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '1.000', '7.500');
  });

  it('BT1 a background drag replaces an all-tracks range, clears the selection once and pointercancel keeps the last range', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const { api } = mounted;
    act(() => live(api).setTimeRange({ start: 2, end: 3, trackId: 'stems', allTracks: true }));
    expect(mounted.laneRanges('sfx')).toHaveLength(1);
    dragBackground(mounted, xFor(1), [xFor(4)], { release: false });
    expect(live(api).range).toEqual({ start: 1, end: 4, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '1.000', '4.000');
    act(() => live(api).selectClip('m1'));
    fireEvent.pointerMove(window, { pointerId: 1, clientX: xFor(7.5), clientY: MUSIC_LANE_Y });
    expect(live(api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    expect(live(api).selectedClipIds).toEqual(['m1']);
    fireEvent.pointerCancel(window, { pointerId: 1, clientX: xFor(7.5), clientY: MUSIC_LANE_Y });
    expect(live(api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    fireEvent.pointerMove(window, { pointerId: 1, clientX: xFor(9), clientY: MUSIC_LANE_Y });
    expect(live(api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    expect(
      mounted.view.getByTestId('arrangement-insert-marker').getAttribute('data-insert-marker'),
    ).toBe('0.000');
  });

  it('BT2 Snap on 1 s snaps the ends; Alt keeps them free; both ends on one line is no range', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    pressKey({ key: 'g', code: 'KeyG' });
    dragBackground(mounted, xFor(1.3), [xFor(7.6)]);
    expect(live(mounted.api).range).toEqual({ start: 1, end: 8, trackId: 'music' });
    dragBackground(mounted, xFor(1.3), [xFor(7.6)], { altKey: true });
    expect(live(mounted.api).range).toEqual({ start: 1.3, end: 7.6, trackId: 'music' });
    dragBackground(mounted, xFor(1.1), [xFor(1.3)]);
    expect(live(mounted.api).range).toBeNull();
    expect(mounted.laneRanges('music')).toHaveLength(0);
  });

  it('BT2 Snap on also snaps a background range end to a marker', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS, [
      { id: 'mk1', time: 5.45, name: 'Hit' },
    ]);
    pressKey({ key: 'g', code: 'KeyG' });
    dragBackground(mounted, xFor(1), [xFor(5.4)]);
    expect(live(mounted.api).range).toEqual({ start: 1, end: 5.45, trackId: 'music' });
  });

  it('BT3 a drag from the background across a clip and a gap into a second clip spans both; Delete leaves the gap', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragBackground(mounted, xFor(5), [xFor(15), xFor(22), xFor(27)]);
    expect(live(mounted.api).range).toEqual({ start: 5, end: 27, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '5.000', '27.000');
    pressKey({ key: 'Delete', code: 'Delete' });
    expect(live(mounted.api).arrangement.clips).toEqual([
      {
        id: expect.any(String),
        trackId: 'music',
        sourceId: 'song',
        start: 27,
        sourceOffset: 2,
        duration: 3,
      },
    ]);
    expect(statusText(mounted)).toBe(
      studioEditStrings.deleteRangeDone({
        start: '00:05.000',
        end: '00:27.000',
        lanes: studioEditStrings.onTrack('Music'),
      }),
    );
  });

  it('BT3 a range over empty space only, then Delete, changes nothing', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragBackground(mounted, xFor(1), [xFor(7.5)]);
    const before = live(mounted.api).arrangement;
    pressKey({ key: 'Delete', code: 'Delete' });
    expect(live(mounted.api).arrangement).toBe(before);
    expect(statusText(mounted)).toBe(studioEditStrings.deleteRangeEmpty);
  });

  it.each([
    ['ctrlKey', { ctrlKey: true }],
    ['metaKey', { metaKey: true }],
  ] as const)('BT4 %s+drag is the marquee: it adds to the earlier selection and clears the range', async (_name, modifier) => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const { api, view } = mounted;
    act(() => live(api).setTimeRange({ start: 1, end: 2, trackId: 'music' }));
    act(() => live(api).selectClip('m1'));
    dragBackground(mounted, HEADER_WIDTH_PX + xFor(24), [HEADER_WIDTH_PX + xFor(28)], {
      ...modifier,
      trackId: 'stems',
      startY: STEMS_LANE_Y,
      moveY: MUSIC_LANE_Y,
      release: false,
    });
    expect(view.getByTestId('arrangement-marquee')).toBeTruthy();
    expect(live(api).range).toBeNull();
    expect([...live(api).selectedClipIds].sort()).toEqual(['m1', 'm2']);
    fireEvent.pointerUp(window, {
      pointerId: 1,
      clientX: HEADER_WIDTH_PX + xFor(28),
      clientY: MUSIC_LANE_Y,
    });
    expect(view.queryByTestId('arrangement-marquee')).toBeNull();
    expect(live(api).range).toBeNull();
    expect([...live(api).selectedClipIds].sort()).toEqual(['m1', 'm2']);
    expect(view.queryAllByTestId('arrangement-lane-range')).toHaveLength(0);
  });

  it('BT5 a click under the threshold sets the insert marker and makes no range', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    act(() => live(mounted.api).setTimeRange({ start: 1, end: 2, trackId: 'music' }));
    dragBackground(mounted, xFor(3), [xFor(3) + 2]);
    expect(live(mounted.api).range).toBeNull();
    const marker = mounted.view.getByTestId('arrangement-insert-marker');
    expect(marker.getAttribute('data-insert-marker')).toBe('3.000');
    expect(marker.hidden).toBe(false);
    expect(mounted.view.queryAllByTestId('arrangement-lane-range')).toHaveLength(0);
  });

  it('BT6 an in-clip drag continues past the clip start and end, bounded by 0 and the lane extent', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const extent = laneExtentSeconds(mounted);
    expect(extent).toBeGreaterThan(30);
    const from = xFor(2);
    dragClipBody(mounted, 'm1', from, [from - xFor(4)]);
    expect(live(mounted.api).range).toEqual({ start: 8, end: 12, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '8.000', '12.000');
    dragClipBody(mounted, 'm1', from, [from + xFor(10)]);
    expect(live(mounted.api).range).toEqual({ start: 12, end: 22, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '12.000', '22.000');
    dragClipBody(mounted, 'm1', from, [from - xFor(20)]);
    expect(live(mounted.api).range).toEqual({ start: 0, end: 12, trackId: 'music' });
    dragClipBody(mounted, 'm1', from, [from + xFor(1000)]);
    expect(live(mounted.api).range).toEqual({ start: 12, end: extent, trackId: 'music' });
  });

  it('BT6 a background drag is bounded by 0 and the lane extent', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const extent = laneExtentSeconds(mounted);
    dragBackground(mounted, xFor(4), [xFor(1000)]);
    expect(live(mounted.api).range).toEqual({ start: 4, end: extent, trackId: 'music' });
    dragBackground(mounted, xFor(4), [-xFor(10)]);
    expect(live(mounted.api).range).toEqual({ start: 0, end: 4, trackId: 'music' });
  });

  it('BT7 a range inside one clip keeps the in-clip overlay and renders no lane overlay', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragClipBody(mounted, 'm1', xFor(2), [xFor(5)]);
    expect(live(mounted.api).range).toEqual({ start: 12, end: 15, trackId: 'music' });
    expect(mounted.view.container.querySelectorAll('[data-range-start]')).toHaveLength(0);
    expect(mounted.clipParts('m1').root.textContent).toContain('00:12.000 – 00:15.000');
  });

  it('BT8 a background drag while playing neither stops nor moves playback', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    const { api } = mounted;
    pressKey({ key: ' ', code: 'Space' });
    expect(live(api).isPlaying).toBe(true);
    fakeClock.base -= 1;
    act(() => api.position?.setPosition(live(api).getLivePosition()));
    expect(positionOf(api)).toBeGreaterThan(0.5);
    const before = live(api).getLivePosition();
    dragBackground(mounted, xFor(5), [xFor(9)]);
    expect(live(api).range).toEqual({ start: 5, end: 9, trackId: 'music' });
    expect(live(api).isPlaying).toBe(true);
    expect(live(api).getLivePosition()).toBeGreaterThanOrEqual(before);
    expect(live(api).getLivePosition()).toBeLessThan(before + 0.5);
  });

  it('BT9 a vertical-only drag makes no zero-length range, and a drag back to its start removes the range', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragBackground(mounted, xFor(3), [xFor(3)], { moveY: STEMS_LANE_Y });
    expect(live(mounted.api).range).toBeNull();
    expect(mounted.view.queryAllByTestId('arrangement-lane-range')).toHaveLength(0);
    dragBackground(mounted, xFor(3), [xFor(6), xFor(3)]);
    expect(live(mounted.api).range).toBeNull();
    expect(mounted.view.queryAllByTestId('arrangement-lane-range')).toHaveLength(0);
  });

  it('BT9 a drag that ends over another lane keeps the range on the origin lane', async () => {
    const mounted = await mountBackgroundWorkspace(TWO_MUSIC_CLIPS);
    dragBackground(mounted, xFor(1), [xFor(4), xFor(7.5)], { moveY: STEMS_LANE_Y });
    expect(live(mounted.api).range).toEqual({ start: 1, end: 7.5, trackId: 'music' });
    expectOnlyMusicOverlay(mounted, '1.000', '7.500');
  });
});
