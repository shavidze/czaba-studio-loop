+    if (scrollerElement === null || header === null) return null;
+    const laneLeft =
+      scrollerElement.getBoundingClientRect().left + header.getBoundingClientRect().width;
+    return element.getBoundingClientRect().right - laneLeft;
+  });

+    return null;
+  }
