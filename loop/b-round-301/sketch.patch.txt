+function listSourceFiles(dir: string): string[] {
+  const files: string[] = [];
+  for (const entry of readdirSync(dir)) {
+    const path = join(dir, entry);
+    if (statSync(path).isDirectory()) {
+      if (entry !== '__tests__') {
+        files.push(...listSourceFiles(path));
+      }
+    } else if (/\.(ts|tsx)$/.test(entry)) {
+      files.push(path);
+    }
+  }
+  return files;
