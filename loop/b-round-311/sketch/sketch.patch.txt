+function signal(seed: number, length: number): Float32Array<ArrayBuffer> {
+  const data = new Float32Array(length);
+  let x = seed;
+  for (let i = 0; i < length; i++) {
+    x = (x * 1103515245 + 12345) % 2147483648;
+    data[i] = (x / 2147483648) * 2 - 1;
+  }

+function streamed(channels: Float32Array[], sampleRate: number, chunk: number): Float32Array {
+  const length = channels[0]?.length ?? 0;
+  const peaks = new Float32Array(peakBucketCount(length, sampleRate, PEAK_BUCKETS_PER_SECOND));
+  for (let start = 0; start < length; start += chunk) {
+    const size = Math.min(chunk, length - start);
+    accumulatePeakChunk(
+      peaks,
+      channels.map((channel) => {
+        const padded = new Float32Array(chunk).fill(1);
+        padded.set(channel.subarray(start, start + size));
+        return padded;
+      }),
+      start,
+      size,
+      sampleRate,
+      PEAK_BUCKETS_PER_SECOND,
+    );
+  }
+  return peaks;
+}

+type PeaksWorkerReply =

+export function readPeaksWorkerReply(data: unknown): PeaksWorkerReply | null {
+  if (typeof data !== 'object' || data === null || !('type' in data) || !('jobId' in data)) {
+    return null;
+  }
+  const { type, jobId } = data;
+  if (typeof jobId !== 'number') {
+    return null;
+  }
+  if (type === 'ack' && 'channels' in data && isChannelList(data.channels)) {
+    return { type, jobId, channels: data.channels };
+  }
+  if (type === 'done' && 'peaks' in data && data.peaks instanceof Float32Array) {
+    return { type, jobId, peaks: data.peaks };
+  }
+  return null;

+      const channels = takeChannels(job.buffer);
+      channels.forEach((channel, index) => {
+        job.buffer.copyFromChannel(channel, index, start);
+      });

+      job.inFlight += 1;
+      post(

+        channels.map((channel) => channel.buffer),
+      );
+    }
+    if (job.next >= job.buffer.length && job.inFlight === 0) {
+      post({ type: 'end', jobId: job.jobId });
+    }
+  };
+
+  const startNext = () => {
+    const job = queue.shift();
+    if (!job) {
+      window.clearTimeout(stallTimer);
+      return;
+    }
+    active = job;
+    pool.length = 0;
+    post({
+      type: 'begin',
+      jobId: job.jobId,
+      length: job.buffer.length,
+      sampleRate: job.buffer.sampleRate,
+      bucketsPerSecond: PEAK_BUCKETS_PER_SECOND,
+    });
+    armStall();
+    pump();
+  };

+    const reply = readPeaksWorkerReply(event.data);
+    const job = active;
+    if (!reply || !job || reply.jobId !== job.jobId) {
+      return;
+    }
+    armStall();
+    if (reply.type === 'ack') {
+      job.inFlight -= 1;
+      pool.push(reply.channels);
+      pump();
+      return;
+    }
+    active = null;
+    if (!job.signal.aborted) {
+      job.onPeaks(reply.peaks);
+    }
+    startNext();
+  };
+
+  const ensureWorker = (): boolean => {
+    if (worker) {
+      return true;
+    }
+    if (workerFailed || typeof Worker === 'undefined') {
+      return false;
+    }
+    try {
+      worker = new Worker(new URL('./peaks.worker.ts', import.meta.url), { type: 'module' });
+    } catch {
+      workerFailed = true;
+      return false;
+    }
+    worker.onmessage = handleReply;
+    worker.onerror = failWorker;
+    worker.onmessageerror = failWorker;
+    return true;
+  };
+
+  return {
+    request(buffer, signal, onPeaks) {
+      const job: PeaksJob = { jobId: nextJobId++, buffer, signal, onPeaks, next: 0, inFlight: 0 };
+      if (buffer.length === 0 || !ensureWorker()) {
+        deliverOnMainThread(job);
+        return;
+      }
+      queue.push(job);
+      if (!active) {
+        startNext();
+      }
+    },
+    dispose() {
+      window.clearTimeout(stallTimer);
+      queue.length = 0;
+      active = null;
+      pool.length = 0;
+      if (worker) {
+        worker.onmessage = null;
+        worker.onerror = null;
+        worker.onmessageerror = null;
+        worker.terminate();
+        worker = null;
+      }
+    },
+  };
+}

+export function accumulatePeakChunk(
+  peaks: Float32Array,
+  chunk: readonly Float32Array[],
+  chunkStart: number,
+  chunkLength: number,
+  sampleRate: number,
+  bucketsPerSecond: number,
+): void {
+  if (chunkLength <= 0 || peaks.length === 0) {
+    return;
+  }
+  const samplesPerBucket = sampleRate / bucketsPerSecond;
+  const chunkEnd = chunkStart + chunkLength;
+  let bucket = Math.max(0, Math.floor(chunkStart / samplesPerBucket) - 1);
+  while (bucket < peaks.length && Math.floor((bucket + 1) * samplesPerBucket) <= chunkStart) {
+    bucket += 1;
+  }
+  for (; bucket < peaks.length; bucket++) {
+    const bucketFrom = Math.floor(bucket * samplesPerBucket);
+    if (bucketFrom >= chunkEnd) {
+      break;
+    }
+    const from = Math.max(bucketFrom, chunkStart) - chunkStart;
+    const to = Math.min(Math.floor((bucket + 1) * samplesPerBucket), chunkEnd) - chunkStart;
+    let max = peaks[bucket];
+    for (const channel of chunk) {
+      for (let i = from; i < to; i++) {
+        const value = Math.abs(channel[i]);
+        if (value > max) {
+          max = value;
+        }
+      }
+    }
+    peaks[bucket] = max;
+  }

+  | { type: 'ack'; jobId: number; channels: Float32Array<ArrayBuffer>[] }
+  | { type: 'done'; jobId: number; peaks: Float32Array<ArrayBuffer> };

+interface PeaksWorkerJob {
+  jobId: number;
+  peaks: Float32Array<ArrayBuffer>;
+  sampleRate: number;
+  bucketsPerSecond: number;

+export interface PeaksWorkerState {
+  job: PeaksWorkerJob | null;
+}

+export function handlePeaksWorkerRequest(
+  state: PeaksWorkerState,
+  request: PeaksWorkerRequest,
+  post: (response: PeaksWorkerResponse, transfer: ArrayBuffer[]) => void,
+): void {
+  if (request.type === 'begin') {
+    state.job = {
+      jobId: request.jobId,
+      peaks: new Float32Array(
+        peakBucketCount(request.length, request.sampleRate, request.bucketsPerSecond),
+      ),
+      sampleRate: request.sampleRate,
+      bucketsPerSecond: request.bucketsPerSecond,
+    };
+    return;
+  }
+  const job = state.job;
+  if (request.type === 'chunk') {
+    if (job && job.jobId === request.jobId) {
+      accumulatePeakChunk(
+        job.peaks,
+        request.channels,
+        request.start,
+        request.length,
+        job.sampleRate,
+        job.bucketsPerSecond,
+      );
+    }
+    post(
+      { type: 'ack', jobId: request.jobId, channels: request.channels },
+      request.channels.map((channel) => channel.buffer),
+    );
+    return;
+  }
+  if (!job || job.jobId !== request.jobId) {
+    return;
+  }
+  state.job = null;
+  if (request.type === 'end') {
