class FakeAudioBuffer {
  readonly length: number;
  readonly numberOfChannels: number;
  readonly sampleRate: number;
  constructor(options: { length: number; numberOfChannels: number; sampleRate: number }) {
    this.length = options.length;
    this.numberOfChannels = options.numberOfChannels;
    this.sampleRate = options.sampleRate;
  }
  get duration() {
    return this.length / this.sampleRate;
  }
  getChannelData() {
    return new Float32Array(this.length);
  }
}

const PX_PER_SECOND = 40;
const LANE_HEIGHT_PX = 80;
const press = { button: 0, buttons: 1 };

interface SheetApiRef {
  current: ArrangementContextValue | null;
}

function SheetProbe({ api }: { api: SheetApiRef }) {
  api.current = useArrangement();
  return null;
}

function live(api: SheetApiRef): ArrangementContextValue {
  if (!api.current) {
    throw new Error('provider not mounted');
  }
  return api.current;
}

function stubCapture(element: Element) {
  Object.defineProperty(element, 'setPointerCapture', { configurable: true, value: () => {} });
  Object.defineProperty(element, 'hasPointerCapture', { configurable: true, value: () => false });
  Object.defineProperty(element, 'releasePointerCapture', { configurable: true, value: () => {} });
}

async function mountSheetWorkspace() {
  const api: SheetApiRef = { current: null };
  const view = render(
    <ArrangementProvider collection={sheetCollection} initialArrangement={sheetArrangement()}>
      <StudioWorkspace />
      <SheetProbe api={api} />
    </ArrangementProvider>,
  );
  await act(async () => {
    await Promise.resolve();
    await Promise.resolve();
  });
  act(() => live(api).setPxPerSecond(PX_PER_SECOND));
  const rowTops = new Map<string, number>();
  view.container.querySelectorAll<HTMLElement>('[data-track-id]').forEach((row, index) => {
    const top = index * LANE_HEIGHT_PX;
    rowTops.set(row.dataset.trackId ?? '', top);
    Object.defineProperty(row, 'getBoundingClientRect', {
      configurable: true,
      value: () => new DOMRect(0, top, 800, LANE_HEIGHT_PX),
    });
  });
  const laneY = (trackId: string) => {
    const top = rowTops.get(trackId);
    if (top === undefined) {
      throw new Error(`lane ${trackId} not rendered`);
    }
    return top + LANE_HEIGHT_PX / 2;
  };
  const clipParts = (clipId: string) => {
    const root = view.container.querySelector<HTMLElement>(`[data-clip-id="${clipId}"]`);
    const header = root?.firstElementChild;
    if (!root || !(header instanceof HTMLElement)) {
      throw new Error(`clip ${clipId} not rendered`);
    }
    stubCapture(root);
    return { root, header };
  };
  return { api, view, laneY, clipParts };
}

const TOGGLE = { key: 'o', code: 'KeyO', ctrlKey: true, altKey: true };

function status(view: Awaited<ReturnType<typeof mountSheetWorkspace>>['view']): string {
  return view.getByTestId('arrangement-status').textContent ?? '';
}

describe('Overview empty state and Ctrl+Alt+O (reference sketch)', () => {
  it('REF1: toggles, persists, announces and ignores AltGr, Shift, repeat and text fields', async () => {
    const { view } = await mountSheetWorkspace();
    expect(view.getByTestId('arrangement-overview').getAttribute('data-overview-state')).toBe('full');
    expect(fireEvent.keyDown(document.body, { ...TOGGLE, modifierAltGraph: true })).toBe(true);
    expect(view.queryByTestId('arrangement-overview')).not.toBeNull();
    fireEvent.keyDown(document.body, { ...TOGGLE, shiftKey: true });
    expect(view.queryByTestId('arrangement-overview')).not.toBeNull();
    fireEvent.keyDown(document.body, { ...TOGGLE, repeat: true });
    expect(view.queryByTestId('arrangement-overview')).not.toBeNull();
    const search = view.getByTestId('arrangement-library-search');
    fireEvent.keyDown(search, TOGGLE);
    expect(view.queryByTestId('arrangement-overview')).not.toBeNull();
    expect(fireEvent.keyDown(document.body, TOGGLE)).toBe(false);
    expect(view.queryByTestId('arrangement-overview')).toBeNull();
    expect(view.getByTestId('arrangement-overview-hidden-slot').getAttribute('aria-hidden')).toBe('true');
    expect(window.localStorage.getItem(OVERVIEW_HIDDEN_STORAGE_KEY)).toBe('1');
    expect(status(view)).toBe('Overview hidden. Ctrl+Alt+O shows it again.');
    fireEvent.keyDown(document.body, { ...TOGGLE, ctrlKey: false, metaKey: true });
    expect(view.queryByTestId('arrangement-overview')).not.toBeNull();
    expect(window.localStorage.getItem(OVERVIEW_HIDDEN_STORAGE_KEY)).toBeNull();
    expect(status(view)).toBe('Overview shown. Ctrl+Alt+O hides it.');
  });

  it('REF2: empty project shows the quiet state; the first clip shows the full view; undo returns', async () => {
    const { api, view } = await mountSheetWorkspace();
    act(() => {
      for (const clip of live(api).arrangement.clips) {
        live(api).selectClip(clip.id);
      }
    });
    expect(view.getByTestId('arrangement-overview').getAttribute('data-overview-state')).toBe('full');
    expect(view.queryByTestId('arrangement-overview-baseline')).toBeNull();
  });
});
