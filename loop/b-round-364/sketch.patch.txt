+export function createCachedSourceFetch({
+  store,
+  origin,
+  fetchSource = plainSourceFetch,
+  now = () => Date.now(),
+  estimateQuota = browserQuota,
+}: CachedSourceFetchDeps): CachedSourceFetch {
+  let disabled = store === null;
+  let sweepPromise: Promise<void> | null = null;
+  const writes = new Map<string, Promise<void>>();
+
+  const storeBytes = async (
+    activeStore: StemCacheStore,
+    record: StemCacheRecord,
+    bytes: ArrayBuffer,
+  ): Promise<void> => {
+    sweepPromise ??= activeStore.sweep(now()).catch(() => undefined);
+    await sweepPromise;
+    const victims = selectStemCacheEvictions(
+      await activeStore.list(),
+      record.key,
+      record.size,
+      stemCacheBudgetBytes(await estimateQuota()),
+    );
+    if (victims === null) {
+      return;
+    }
+    for (const victim of victims) {
+      await activeStore.delete(victim.key);
+    }
+    try {
+      await activeStore.put(record, bytes);
+    } catch (error) {
+      if (!isQuotaError(error)) {
+        throw error;
+      }
+      const oldest = (await activeStore.list())
+        .filter((candidate) => candidate.key !== record.key)
+        .sort((first, second) => first.lastAccessedAt - second.lastAccessedAt)[0];
+      if (oldest !== undefined) {
+        await activeStore.delete(oldest.key);
+      }
+      await activeStore.put(record, bytes);
+    }
+  };
+
+  const write = async (
+    activeStore: StemCacheStore,
+    key: string,
+    response: Response,
+    signal: AbortSignal | null | undefined,
+  ): Promise<void> => {
+    let bytes: ArrayBuffer;
+    try {
+      bytes = await response.arrayBuffer();
+    } catch {
+      return;
+    }
+    if (signal?.aborted || disabled) {
+      return;
+    }
+    const validators = readStemCacheValidators(response.headers);
+    if (validators === null) {
+      return;
+    }
+    const at = now();
+    const record: StemCacheRecord = {
+      key,
+      fileName: `${at.toString(36)}-${Math.random().toString(36).slice(2)}`,
+      etag: validators.etag,
+      lastModified: validators.lastModified,
+      contentType: response.headers.get('content-type'),
+      size: bytes.byteLength,
+      storedAt: at,
+      lastAccessedAt: at,
+      schemaVersion: STEM_CACHE_SCHEMA_VERSION,
+    };
+    try {
+      await storeBytes(activeStore, record, bytes);
+    } catch {
+      disabled = true;
+    }
+  };
+
+  const isStorable = (response: Response): boolean =>
+    response.status === 200 && readStemCacheValidators(response.headers) !== null;
+
+  const writeInBackground = (key: string, response: Response, init: RequestInit): void => {
+    if (store === null || disabled || writes.has(key) || !isStorable(response)) {
+      return;
+    }
+    const pending = write(store, key, response.clone(), init.signal).finally(() => {
+      writes.delete(key);
+    });
+    writes.set(key, pending);
+  };
+
+  const cachedFetch: SourceFetch = async (url, init) => {
+    const key = disabled || store === null ? null : stemCacheKey(url, origin);
+    if (key === null || store === null) {
+      return fetchSource(url, init);
+    }
+    let entry: StemCacheEntry | null;
+    try {
+      entry = await store.get(key);
+    } catch {
+      entry = null;
+    }
+    if (entry === null) {
+      const response = await fetchSource(url, init);
+      writeInBackground(key, response, init);
+      return response;
+    }
+    let response: Response;
+    try {
+      response = await fetchSource(url, {
+        ...init,
+        cache: 'no-store',
+        headers: stemCacheConditionalHeaders(entry.record),
+      });
+    } catch (error) {
+      if (init.signal?.aborted) {
+        throw error;
+      }
+      return cachedResponse(entry);
+    }
+    if (response.status === 304) {
+      await store.touch(key, now()).catch(() => undefined);
+      return cachedResponse(entry);
+    }
+    if (response.status === 404 || response.status === 410) {
+      await store.delete(key).catch(() => undefined);
+      return response;
+    }
+    if (response.status >= 500) {
+      return cachedResponse(entry);
+    }
+    if (response.ok && !isStorable(response)) {
+      await store.delete(key).catch(() => undefined);
+      return response;
+    }
+    writeInBackground(key, response, init);
+    return response;
+  };
+
+  return {
+    fetchSource: cachedFetch,
+    idle: async () => {
+      await Promise.all([...writes.values()]);
+    },
+  };

+export const STEM_CACHE_DB_NAME = 'czaba-stem-cache-v2';
+export const STEM_CACHE_SCHEMA_VERSION = 1;
+export const STEM_CACHE_MAX_BUDGET_BYTES = 256 * 1024 * 1024;
+export const STEM_CACHE_QUOTA_SHARE = 0.1;

+export interface StemCacheRecord {
+  key: string;
+  fileName: string;
+  etag: string | null;
+  lastModified: string | null;
+  contentType: string | null;
+  size: number;
+  storedAt: number;

+  schemaVersion: number;

+export interface StemCacheValidators {
+  etag: string | null;
+  lastModified: string | null;

+export function stemCacheKey(url: string, origin: string): string | null {
+  let parsed: URL;
+  try {
+    parsed = new URL(url, origin);
+  } catch {
+    return null;
+  }
+  if (parsed.origin !== origin || parsed.search !== '' || parsed.hash !== '') {
+    return null;
+  }
+  return parsed.pathname;
+}

+export function readStemCacheValidators(headers: Headers): StemCacheValidators | null {
+  const cacheControl = (headers.get('cache-control') ?? '').toLowerCase();
+  if (cacheControl.includes('no-store')) {
+    return null;
+  }
+  const vary = (headers.get('vary') ?? '')
+    .split(',')
+    .map((part) => part.trim().toLowerCase())
+    .filter((part) => part.length > 0 && part !== 'accept-encoding' && part !== 'origin');
+  if (vary.length > 0) {
+    return null;
+  }
+  const etag = headers.get('etag');
+  const lastModified = headers.get('last-modified');
+  if (etag === null && lastModified === null) {
+    return null;
+  }
+  return { etag, lastModified };
+}

+export function stemCacheConditionalHeaders(record: StemCacheRecord): Headers {
+  const headers = new Headers();
+  if (record.etag !== null) {
+    headers.set('If-None-Match', record.etag);
+  }
+  if (record.lastModified !== null) {
+    headers.set('If-Modified-Since', record.lastModified);
+  }
+  return headers;
+}

+export function stemCacheBudgetBytes(quota: number | undefined): number {
+  if (quota === undefined || !Number.isFinite(quota) || quota <= 0) {
+    return 0;
+  }
+  return Math.min(STEM_CACHE_MAX_BUDGET_BYTES, Math.floor(quota * STEM_CACHE_QUOTA_SHARE));
+}

+export function selectStemCacheEvictions(
+  records: readonly StemCacheRecord[],
+  incomingKey: string,
+  incomingSize: number,
+  budget: number,
+): StemCacheRecord[] | null {
+  if (incomingSize <= 0 || incomingSize > budget / 4) {
+    return null;
+  }
+  const others = records
+    .filter((record) => record.key !== incomingKey)
+    .sort((first, second) => first.lastAccessedAt - second.lastAccessedAt);
+  let used = others.reduce((total, record) => total + record.size, 0);
+  const victims: StemCacheRecord[] = [];
+  for (const record of others) {
+    if (used + incomingSize <= budget) {
+      break;
+    }
+    victims.push(record);
+    used -= record.size;
+  }
+  return victims;
