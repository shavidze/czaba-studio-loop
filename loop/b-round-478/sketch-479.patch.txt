+function openingTags(source: string, tagName: string): string[] {
+  const tags: string[] = [];
+  const pattern = new RegExp(`<${tagName}(?=[\\s>])`, 'g');
+  for (let match = pattern.exec(source); match; match = pattern.exec(source)) {
+    let depth = 0;
+    let quote = '';
+    let end = match.index + match[0].length;
+    for (; end < source.length; end++) {
+      const char = source[end];
+      if (quote) {
+        if (char === quote) quote = '';
+      } else if (char === '"' || char === "'" || char === '`') {
+        quote = char;
+      } else if (char === '{') {
+        depth++;
+      } else if (char === '}') {
+        depth--;
+      } else if (char === '>' && depth === 0) {
+        break;
+      }
+    }
+    tags.push(source.slice(match.index, end + 1));
+  }
+  return tags;
+}

+import type { MouseEvent } from 'react';

+function isPlainPrimaryActivation(event: MouseEvent<HTMLAnchorElement>): boolean {
+  return (
+    !event.defaultPrevented &&
+    event.button === 0 &&
+    !event.metaKey &&
+    !event.altKey &&
+    !event.ctrlKey &&
+    !event.shiftKey
+  );
+}
+
+export function releaseLinkFocus(event: MouseEvent<HTMLAnchorElement>): void {
+  if (!isPlainPrimaryActivation(event)) return;
+  const { body } = event.currentTarget.ownerDocument;
+  const hadTabIndex = body.hasAttribute('tabindex');
+  if (!hadTabIndex) body.tabIndex = -1;
+  body.focus({ preventScroll: true });
+  if (!hadTabIndex) body.removeAttribute('tabindex');
+}
