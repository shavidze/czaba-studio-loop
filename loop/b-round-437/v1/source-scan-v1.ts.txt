import { readdirSync, readFileSync, statSync } from 'node:fs';
import { join, relative, resolve } from 'node:path';

export const STORE_ROOT = resolve(__dirname, '../../../../..');
export const ARRANGEMENT_COMPONENTS_DIR = resolve(STORE_ROOT, 'app/components/arrangement');

export interface SourceFile {
  path: string;
  text: string;
}

export interface JsxOpeningTag {
  path: string;
  line: number;
  tag: string;
}

const SKIPPED_DIRECTORIES = new Set(['node_modules', '__tests__']);
const TEST_FILE = /\.(test|spec)\.(ts|tsx)$/;
const IMPORT_STATEMENT = /^import\s[^;]*;|^import\s*['"][^'"]+['"];?/gm;

export function listSourceFiles(dir: string, extensions: readonly string[] = ['.tsx']): string[] {
  const files: string[] = [];
  for (const entry of readdirSync(dir).sort()) {
    const path = join(dir, entry);
    if (statSync(path).isDirectory()) {
      if (!SKIPPED_DIRECTORIES.has(entry)) {
        files.push(...listSourceFiles(path, extensions));
      }
    } else if (
      extensions.some((extension) => entry.endsWith(extension)) &&
      !TEST_FILE.test(entry)
    ) {
      files.push(path);
    }
  }
  return files;
}

export function readSourceFiles(
  dir: string = ARRANGEMENT_COMPONENTS_DIR,
  extensions: readonly string[] = ['.tsx'],
): SourceFile[] {
  return listSourceFiles(dir, extensions).map((file) => ({
    path: relative(dir, file).split('\\').join('/'),
    text: readFileSync(file, 'utf8'),
  }));
}

function skipQuoted(source: string, from: number, quote: string): number {
  for (let index = from + 1; index < source.length; index++) {
    if (source[index] === '\\') {
      index++;
    } else if (source[index] === quote) {
      return index + 1;
    }
  }
  return -1;
}

interface JsxTagWalk {
  end: number;
  skeleton: string;
}

function walkJsxTag(source: string, from: number): JsxTagWalk | null {
  const stack: ('brace' | 'template')[] = [];
  let skeleton = '';
  let braceStart = -1;
  let index = from;
  while (index < source.length) {
    const char = source[index];
    const next = source[index + 1];
    if (stack[stack.length - 1] === 'template') {
      if (char === '\\') {
        index += 2;
      } else if (char === '`') {
        stack.pop();
        index++;
      } else if (char === '$' && next === '{') {
        stack.push('brace');
        index += 2;
      } else {
        index++;
      }
      continue;
    }
    if (char === '"' || char === "'") {
      const close = skipQuoted(source, index, char);
      if (close === -1) {
        return null;
      }
      if (stack.length === 0) {
        skeleton += '""';
      }
      index = close;
    } else if (char === '`') {
      stack.push('template');
      index++;
    } else if (char === '/' && next === '/') {
      const lineEnd = source.indexOf('\n', index);
      if (lineEnd === -1) {
        return null;
      }
      index = lineEnd + 1;
    } else if (char === '/' && next === '*') {
      const commentEnd = source.indexOf('*/', index + 2);
      if (commentEnd === -1) {
        return null;
      }
      index = commentEnd + 2;
    } else if (char === '{') {
      if (stack.length === 0) {
        braceStart = index;
      }
      stack.push('brace');
      index++;
    } else if (char === '}') {
      stack.pop();
      if (stack.length === 0 && braceStart !== -1) {
        skeleton += /^\{\s*\.\.\./.test(source.slice(braceStart, index)) ? '{...}' : '{}';
        braceStart = -1;
      }
      index++;
    } else if (char === '>' && stack.length === 0) {
      return { end: index, skeleton: `${skeleton}>` };
    } else {
      if (stack.length === 0) {
        skeleton += char;
      }
      index++;
    }
  }
  return null;
}

export function jsxTagSkeleton(tag: string): string {
  const opener = tag.match(/^<[\w.]+/)?.[0] ?? '';
  const walk = walkJsxTag(tag, opener.length);
  return walk ? `${opener}${walk.skeleton}` : tag;
}

export function jsxOpeningTags(source: string, name: string, path = '<inline>'): JsxOpeningTag[] {
  const tags: JsxOpeningTag[] = [];
  const opener = new RegExp(`<${name}(?=[\\s/>])`, 'g');
  for (let match = opener.exec(source); match; match = opener.exec(source)) {
    const line = source.slice(0, match.index).split('\n').length;
    const end = walkJsxTag(source, match.index + name.length + 1)?.end ?? -1;
    if (end === -1) {
      throw new Error(`${path}:${line}: <${name}> tag has no end the scanner can read`);
    }
    tags.push({ path, line, tag: source.slice(match.index, end + 1) });
    opener.lastIndex = end;
  }
  return tags;
}

export function scanJsxOpeningTags(name: string, files: readonly SourceFile[]): JsxOpeningTag[] {
  return files.flatMap((file) => jsxOpeningTags(file.text, name, file.path));
}

export function jsxAttributeLiteral(tag: string, attribute: string): string | null {
  const pattern = new RegExp(
    `(?<![\\w-])${attribute}=(?:"([^"]*)"|'([^']*)'|\\{\\s*(?:"([^"]*)"|'([^']*)'|\`([^\`$]*)\`)\\s*\\})`,
  );
  const match = tag.match(pattern);
  if (!match) {
    return null;
  }
  return match.slice(1).find((group) => group !== undefined) ?? null;
}

export type JsxTagHazard = 'spread' | 'style';

export function jsxTagHazards(tag: string): JsxTagHazard[] {
  const skeleton = jsxTagSkeleton(tag);
  const hazards: JsxTagHazard[] = [];
  if (skeleton.includes('{...}')) {
    hazards.push('spread');
  }
  if (/(?<![\w-])style=/.test(skeleton)) {
    hazards.push('style');
  }
  return hazards;
}

export function stripImportStatements(source: string): string {
  return source.replace(IMPORT_STATEMENT, '');
}
