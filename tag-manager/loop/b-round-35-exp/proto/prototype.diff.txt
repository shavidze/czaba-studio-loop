+function matchesButtonName(button: HTMLButtonElement, name: string | RegExp): boolean {
+  const text = (button.textContent ?? "").trim();
+  return typeof name === "string" ? text === name : name.test(text);
+}
+
+export function getButtonByText(name: string | RegExp): HTMLButtonElement {
+  const matches = Array.from(document.querySelectorAll("button")).filter((button) => matchesButtonName(button, name));
+  if (matches.length === 0) throw new Error(`button ${String(name)} not found`);
+  if (matches.length > 1) throw new Error(`button ${String(name)} matched ${matches.length} buttons`);
+  return matches[0];
+}
