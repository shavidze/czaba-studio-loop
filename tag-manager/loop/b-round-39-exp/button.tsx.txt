import * as React from 'react';
import { cn } from '~/lib/utils';

interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'ghost' | 'accent' | 'destructive';
  size?: 'sm' | 'md' | 'lg' | 'icon';
}

const variantStyles: Record<NonNullable<ButtonProps['variant']>, string> = {
  primary: 'bg-brand text-brand-foreground enabled:not-aria-disabled:hover:bg-brand-hover',
  secondary:
    'bg-surface text-foreground border border-outline enabled:not-aria-disabled:hover:bg-elevated',
  ghost: 'text-foreground enabled:not-aria-disabled:hover:bg-elevated',
  accent: 'bg-accent text-white enabled:not-aria-disabled:hover:bg-accent/90',
  destructive: 'bg-destructive text-white enabled:not-aria-disabled:hover:bg-destructive/90',
};

const sizeStyles: Record<NonNullable<ButtonProps['size']>, string> = {
  sm: 'px-2.5 py-1 text-xs',
  md: 'px-4 py-2 text-sm',
  lg: 'px-6 py-3 text-base',
  icon: 'p-2',
};

const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ variant = 'primary', size = 'md', className, ...props }, ref) => {
    return (
      <button
        ref={ref}
        type="button"
        className={cn(
          'inline-flex items-center justify-center gap-1.5 rounded-md font-medium',
          'transition-colors duration-150',
          'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand',
          'disabled:opacity-50 disabled:cursor-not-allowed',
          'aria-disabled:opacity-50 aria-disabled:cursor-not-allowed',
          variantStyles[variant],
          sizeStyles[size],
          className,
        )}
        {...props}
      />
    );
  },
);

Button.displayName = 'Button';

export { Button };
export type { ButtonProps };
