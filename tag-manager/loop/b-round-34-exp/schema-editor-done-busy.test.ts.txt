import { act, cleanup, fireEvent, render, screen } from "@testing-library/react";
import { createElement, Fragment } from "react";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { MetadataPanel } from "~/components/metadata-editor/MetadataPanel";
import { SchemaEditor } from "~/components/schema/SchemaEditor";
import { FolderBrowser } from "~/components/sidebar/FolderBrowser";
import { useFileStore } from "~/stores/file-store";
import { useMetadataStore } from "~/stores/metadata-store";
import type { FieldDefinition, WavFileEntry, WavMetadata } from "~/types";
import { resetAllStores } from "../helpers/reset-stores";
import { resetWindowApiMocks } from "../helpers/window-api-mocks";

vi.hoisted(() => {
  const globalScope = globalThis as unknown as { window: { api: unknown }; api: unknown };
  const mockApi = globalScope.window.api;
  globalScope.window = globalThis as unknown as { api: unknown };
  globalScope.api = mockApi;
});

type MetadataByPath = Record<string, WavMetadata>;

interface Deferred<T> {
  promise: Promise<T>;
  resolve: (value: T) => void;
}

const MODIFIED_AT = "2026-10-28T00:00:00.000Z";
const PATH_A = "/fa/a.wav";
const PATH_B = "/fb/b.wav";
const FILE_A: WavFileEntry = { name: "a.wav", path: PATH_A, size: 1, modifiedAt: MODIFIED_AT };
const FILE_B: WavFileEntry = { name: "b.wav", path: PATH_B, size: 1, modifiedAt: MODIFIED_AT };
const BASE_BY_PATH: MetadataByPath = {
  [PATH_A]: { title: "A", category: "ACat", comment: "" },
  [PATH_B]: { title: "B", category: "BCat", comment: "" },
};
const FIELD_NAME_PLACEHOLDER = "Field name (e.g. Client)";

function createDeferred<T>(): Deferred<T> {
  let resolve: (value: T) => void = () => {};
  const promise = new Promise<T>((onResolve) => {
    resolve = onResolve;
  });
  return { promise, resolve };
}

function buttonByText(name: RegExp): HTMLButtonElement {
  const found = Array.from(document.querySelectorAll("button")).find((button) => name.test(button.textContent ?? ""));
  if (!found) throw new Error(`button ${name} not found`);
  return found;
}

function typeSelect(): HTMLSelectElement {
  const select = document.querySelector("select");
  if (!select) throw new Error("type select not found");
  return select;
}

async function clickElement(element: HTMLElement): Promise<void> {
  await act(async () => {
    fireEvent.click(element);
  });
}

async function clickButton(name: RegExp): Promise<void> {
  await clickElement(buttonByText(name));
}

function backButton(): HTMLButtonElement {
  const back = document.querySelector("svg.lucide-arrow-left")?.closest("button");
  if (!back) throw new Error("back button not found");
  return back;
}

async function settle(action: () => void): Promise<void> {
  await act(async () => {
    action();
  });
}

async function waitMs(ms: number): Promise<void> {
  await act(async () => {
    await new Promise((resolve) => setTimeout(resolve, ms));
  });
}

function renderBrowserAndPanel(): void {
  render(createElement(Fragment, null, createElement(FolderBrowser), createElement(MetadataPanel)));
}

function seedFolderA(): void {
  useFileStore.setState({
    folderPath: "/fa",
    files: [FILE_A],
    metadata: { [PATH_A]: { ...BASE_BY_PATH[PATH_A] } },
    selectedPaths: [PATH_A],
  });
}

function readFor(paths: string[], schema: FieldDefinition[]): MetadataByPath {
  const out: MetadataByPath = {};
  for (const path of paths) {
    const meta: WavMetadata = { ...BASE_BY_PATH[path] };
    for (const field of schema) {
      if (!field.standard) meta[field.key] = `${field.key}@${path}`;
    }
    out[path] = meta;
  }
  return out;
}

function fieldNameInput(): HTMLInputElement | null {
  return screen.queryByPlaceholderText(FIELD_NAME_PLACEHOLDER);
}

function typeFieldName(value: string): void {
  fireEvent.change(screen.getByPlaceholderText(FIELD_NAME_PLACEHOLDER), { target: { value } });
}

async function openCustomize(): Promise<void> {
  await clickElement(screen.getByTitle("Customize Fields"));
}

async function addField(name: string): Promise<void> {
  typeFieldName(name);
  await clickButton(/Add Field/);
}

function readsContaining(path: string): number {
  return vi.mocked(window.api.readMetadataBatch).mock.calls.filter(([paths]) => paths.includes(path)).length;
}

async function doneDuringPendingOpenFolderB(): Promise<{ scanB: Deferred<WavFileEntry[]>; done: HTMLElement }> {
  seedFolderA();
  const scanB = createDeferred<WavFileEntry[]>();
  vi.mocked(window.api.openFolderDialog).mockResolvedValueOnce("/fb");
  vi.mocked(window.api.scanFolder).mockReturnValueOnce(scanB.promise);
  vi.mocked(window.api.readMetadataBatch).mockImplementation(async (paths, schema) => readFor(paths, schema));
  renderBrowserAndPanel();
  await openCustomize();
  await clickButton(/Open Folder/);
  await addField("R35note");
  const done = buttonByText(/Done/);
  await clickElement(done);
  return { scanB, done };
}

beforeEach(() => {
  resetAllStores();
  resetWindowApiMocks();
  vi.mocked(window.api.openFolderDialog).mockReset().mockResolvedValue(null);
  vi.mocked(window.api.loadSchema).mockReset().mockResolvedValue([]);
  vi.mocked(window.api.resolveDroppedPaths).mockReset().mockResolvedValue({ folders: [], wavFiles: [] });
});

describe("schema editor Done busy state and latest schema re-read (D-42, D-43)", () => {
  afterEach(() => {
    cleanup();
  });

  it("(b1) RR-R33-01: while Done is pending the button reads Applying... and is disabled, the editing controls are disabled and Back stays enabled", async () => {
    const { scanB, done } = await doneDuringPendingOpenFolderB();

    expect(done).toBeDisabled();
    expect(done).toHaveTextContent("Applying...");
    expect(fieldNameInput()).toBeDisabled();
    expect(buttonByText(/Add Field/)).toBeDisabled();
    expect(screen.getByPlaceholderText("RIFF ID (auto)")).toBeDisabled();
    expect(typeSelect()).toBeDisabled();
    expect(backButton()).toBeEnabled();

    await settle(() => scanB.resolve([FILE_B]));
    await waitMs(20);
    expect(fieldNameInput()).toBeNull();
  });

  it("(b2) QA-R33-02: a second click on the pending Done starts no second schema re-read", async () => {
    const { scanB, done } = await doneDuringPendingOpenFolderB();
    await clickElement(done);
    await settle(() => scanB.resolve([FILE_B]));
    await waitMs(20);

    expect(readsContaining(PATH_B)).toBe(2);
    expect(useFileStore.getState().metadata[PATH_B]?.r35note).toBe(`r35note@${PATH_B}`);
    expect(fieldNameInput()).toBeNull();
  });

  it("(b3) pin with D-41: Done pending, Back, reopen Customize Fields: the new editor's Done reads Done and is enabled, its inputs are enabled", async () => {
    const { scanB } = await doneDuringPendingOpenFolderB();
    await clickElement(backButton());
    await settle(() => useFileStore.getState().setSelectedPaths([PATH_A]));
    await openCustomize();

    const reopenedDone = buttonByText(/^Done$/);
    expect(reopenedDone).toBeEnabled();
    expect(fieldNameInput()).toBeEnabled();

    await settle(() => scanB.resolve([FILE_B]));
    await waitMs(20);
    expect(fieldNameInput()).not.toBeNull();
  });

  it("(b4) review R34 low 1: onSave rejects: the editor stays open, onClose is not called, Done reads Done and is enabled again", async () => {
    const onClose = vi.fn();
    const onSave = vi.fn(() => Promise.reject(new Error("reread failed")));
    render(createElement(SchemaEditor, { onClose, onSave }));
    await clickButton(/Done/);
    await waitMs(20);

    expect(onSave).toHaveBeenCalledTimes(1);
    expect(onClose).not.toHaveBeenCalled();
    const done = buttonByText(/^Done$/);
    expect(done).toBeEnabled();
    expect(fieldNameInput()).toBeEnabled();
  });

  it("(b5) QA-R34-01: Done #1 (field Qone) pending, Back, reopen, add Qtwo, Done #2; #2's read lands first, #1's last: the store keeps #2's result with both fields", async () => {
    seedFolderA();
    const read1 = createDeferred<MetadataByPath>();
    const read2 = createDeferred<MetadataByPath>();
    vi.mocked(window.api.readMetadataBatch).mockReturnValueOnce(read1.promise).mockReturnValueOnce(read2.promise);
    renderBrowserAndPanel();
    await openCustomize();
    await addField("Qone");
    await clickButton(/Done/);
    const schemaAtDone1 = vi.mocked(window.api.readMetadataBatch).mock.calls[0][1];
    await clickElement(backButton());
    await openCustomize();
    await addField("Qtwo");
    await clickButton(/Done/);
    const schemaAtDone2 = vi.mocked(window.api.readMetadataBatch).mock.calls[1][1];

    await settle(() => read2.resolve(readFor([PATH_A], schemaAtDone2)));
    await waitMs(20);
    expect(fieldNameInput()).toBeNull();
    await settle(() => read1.resolve(readFor([PATH_A], schemaAtDone1)));
    await waitMs(20);

    const expected = readFor([PATH_A], schemaAtDone2)[PATH_A];
    expect(expected.qtwo).toBe(`qtwo@${PATH_A}`);
    expect(useFileStore.getState().metadata).toEqual({ [PATH_A]: expected });
    expect(useMetadataStore.getState().editingMetadata).toEqual(expected);
  });

  it("(b6) pin: Done #1 pending, Back, reopen, Done #2; #1's read lands first, #2's last: the store holds #2's result", async () => {
    seedFolderA();
    const read1 = createDeferred<MetadataByPath>();
    const read2 = createDeferred<MetadataByPath>();
    vi.mocked(window.api.readMetadataBatch).mockReturnValueOnce(read1.promise).mockReturnValueOnce(read2.promise);
    renderBrowserAndPanel();
    await openCustomize();
    await addField("Qone");
    await clickButton(/Done/);
    const schemaAtDone1 = vi.mocked(window.api.readMetadataBatch).mock.calls[0][1];
    await clickElement(backButton());
    await openCustomize();
    await addField("Qtwo");
    await clickButton(/Done/);
    const schemaAtDone2 = vi.mocked(window.api.readMetadataBatch).mock.calls[1][1];

    await settle(() => read1.resolve(readFor([PATH_A], schemaAtDone1)));
    await waitMs(20);
    await settle(() => read2.resolve(readFor([PATH_A], schemaAtDone2)));
    await waitMs(20);

    const expected = readFor([PATH_A], schemaAtDone2)[PATH_A];
    expect(useFileStore.getState().metadata).toEqual({ [PATH_A]: expected });
    expect(useMetadataStore.getState().editingMetadata).toEqual(expected);
    expect(fieldNameInput()).toBeNull();
  });
});
