+  release: () => void;
+}
+
+function seedFiles(selectedPaths: string[]): void {
+  useFileStore.setState({
+    files: [FILE_A, FILE_B],
+    metadata: { [PATH_A]: { ...SEED_A }, [PATH_B]: { ...SEED_B } },
+    selectedPaths,
+  });
+}
+
+function schemaHasNote(fields: FieldDefinition[] | undefined): boolean {
+  return (fields ?? []).some((field) => field.key === NOTE_KEY);
+}
+
+function noteFor(fields: FieldDefinition[]): WavMetadata {
+  return schemaHasNote(fields) ? { [NOTE_KEY]: NOTE_FROM_FILE } : {};
+}
+
+function mockSchemaAwareReads(fileA: WavMetadata): void {
+  vi.mocked(window.api.readMetadataBatch).mockImplementation(async (paths, fields) => {
+    const result: Record<string, WavMetadata> = {};
+    for (const path of paths) {
+      if (path === PATH_A) result[path] = { ...fileA, ...noteFor(fields) };
+      if (path === PATH_B) result[path] = { ...SEED_B };
+    }
+    return result;
+  });
+  vi.mocked(window.api.readMetadata).mockImplementation(async (path, fields) =>
+    path === PATH_A ? { ...fileA, ...noteFor(fields) } : { ...SEED_B },
+  );
+}
+
+function deferSingleWrite(): PendingWrite {
+  let resolveWrite: () => void = () => {};
+  vi.mocked(window.api.writeMetadata).mockReturnValueOnce(
+    new Promise<void>((resolve) => {
+      resolveWrite = () => resolve();
+    }),
+  );
+  return { release: () => resolveWrite() };
+}
+
+function deferBatchWrite(result: BatchWriteResult): PendingWrite {
+  let resolveWrite: () => void = () => {};
+  vi.mocked(window.api.writeMetadataBatch).mockReturnValueOnce(
+    new Promise<BatchWriteResult>((resolve) => {
+      resolveWrite = () => resolve(result);
+    }),
+  );
+  return { release: () => resolveWrite() };
+}
+
+function getInputByLabel(labelText: string): HTMLInputElement {
+  const label = screen.getByText(labelText);
+  const input = label.parentElement?.querySelector("input");
+  if (!input) throw new Error(`${labelText} input not found`);
+  return input;
+}
+
+async function typeCategoryAndSave(saveButtonName: RegExp): Promise<void> {
+  fireEvent.change(getInputByLabel("Category (UCS)"), { target: { value: TYPED_CATEGORY } });
+  await act(async () => {
+    fireEvent.click(screen.getByRole("button", { name: saveButtonName }));
+  });
+}
+
+async function addNoteFieldAndDone(): Promise<void> {
+  fireEvent.click(screen.getByTitle("Customize Fields"));
+  fireEvent.change(screen.getByPlaceholderText("Field name (e.g. Client)"), { target: { value: "R24note" } });
+  fireEvent.click(screen.getByRole("button", { name: /Add Field/ }));
+  await act(async () => {
+    fireEvent.click(screen.getByRole("button", { name: /Done/ }));
+  });
+}
+
+async function settle(release: () => void): Promise<void> {
+  await act(async () => {
+    release();
+  });
+}
+
+beforeEach(() => {
+  resetAllStores();
+  vi.mocked(window.api.writeMetadataBatch).mockReset().mockResolvedValue({ succeeded: 0, failed: [] });
+  vi.mocked(window.api.readMetadataBatch).mockReset().mockResolvedValue({});
+  vi.mocked(window.api.writeMetadata).mockReset().mockResolvedValue(undefined);
+  vi.mocked(window.api.readMetadata).mockReset().mockResolvedValue({});
+  vi.mocked(window.api.scanFolder).mockReset().mockResolvedValue([]);
+  vi.mocked(window.api.saveSchema).mockReset().mockResolvedValue(undefined);
+});
+
+afterEach(() => {
+  cleanup();
+  resetAllStores();
+});
+
+describe("MetadataPanel schema change re-reads files (C-R23-01, D-33)", () => {
+  it("(s1) pin: single selection, no save: the form shows the new field read with the new schema", async () => {
+    seedFiles([PATH_A]);
+    mockSchemaAwareReads(SEED_A);
+    render(createElement(MetadataPanel));
+
+    await addNoteFieldAndDone();
+
+    expect(getInputByLabel("R24note").value).toBe(NOTE_FROM_FILE);
+    expect(getInputByLabel("Category (UCS)").value).toBe("ACat");
+    expect(useFileStore.getState().metadata[PATH_A]?.[NOTE_KEY]).toBe(NOTE_FROM_FILE);
+    const batchCalls = vi.mocked(window.api.readMetadataBatch).mock.calls;
+    expect(batchCalls).toHaveLength(1);
+    expect(batchCalls[0]?.[0]).toEqual([PATH_A, PATH_B]);
+    expect(schemaHasNote(batchCalls[0]?.[1])).toBe(true);
+  });
+
+  it("(s2) pin: batch selection, no save: editingMetadata stays empty and the store gets the new field", async () => {
+    seedFiles([PATH_A, PATH_B]);
+    mockSchemaAwareReads(SEED_A);
+    render(createElement(MetadataPanel));
+
+    await addNoteFieldAndDone();
+
+    expect(useMetadataStore.getState().editingMetadata).toEqual({});
+    expect(useFileStore.getState().metadata[PATH_A]?.[NOTE_KEY]).toBe(NOTE_FROM_FILE);
+    expect(useFileStore.getState().metadata[PATH_B]).toEqual(SEED_B);
+  });
+
+  it("(s3) B-R23-01: a rejected schema re-read keeps the store entries and the form", async () => {
+    seedFiles([PATH_A]);
+    vi.mocked(window.api.readMetadataBatch).mockRejectedValue(new Error("read down"));
+    render(createElement(MetadataPanel));
+
+    await addNoteFieldAndDone();
+
+    expect(useMetadataStore.getState().toastMessage).toBe("Failed to read 2 files");
+    expect(useFileStore.getState().metadata).toEqual({ [PATH_A]: SEED_A, [PATH_B]: SEED_B });
+    expect(getInputByLabel("Category (UCS)").value).toBe("ACat");
+  });
+
+  it("(s4) A-R21-03: single save pending while the schema changes re-reads with the current schema", async () => {
+    seedFiles([PATH_A]);
+    mockSchemaAwareReads({ ...SEED_A, category: TYPED_CATEGORY });
+    const pending = deferSingleWrite();
+    render(createElement(MetadataPanel));
+
+    await typeCategoryAndSave(/Save Tags/);
+    await addNoteFieldAndDone();
+    await settle(pending.release);
+
+    expect(useMetadataStore.getState().toastMessage).toBe("Tags saved");
+    expect(getInputByLabel("Category (UCS)").value).toBe(TYPED_CATEGORY);
+    expect(getInputByLabel("R24note").value).toBe(NOTE_FROM_FILE);
+    expect(useFileStore.getState().metadata[PATH_A]?.[NOTE_KEY]).toBe(NOTE_FROM_FILE);
+    expect(schemaHasNote(vi.mocked(window.api.readMetadata).mock.lastCall?.[1])).toBe(true);
+    const writeCalls = vi.mocked(window.api.writeMetadata).mock.calls;
+    expect(writeCalls).toHaveLength(1);
+    expect(Array.isArray(writeCalls[0]?.[2])).toBe(true);
+    expect(schemaHasNote(writeCalls[0]?.[2])).toBe(false);
+  });
+
+  it("(s5) A-R21-03: batch save pending while the schema changes re-reads with the current schema", async () => {
+    seedFiles([PATH_A, PATH_B]);
+    mockSchemaAwareReads(SEED_A);
+    const pending = deferBatchWrite(ALL_SUCCEEDED);
+    render(createElement(MetadataPanel));
+
+    await typeCategoryAndSave(/Save to 2 files/);
+    await addNoteFieldAndDone();
+    await settle(pending.release);
+
+    expect(useMetadataStore.getState().toastMessage).toBe("Tags saved to 2 files");
+    expect(useFileStore.getState().metadata[PATH_A]?.[NOTE_KEY]).toBe(NOTE_FROM_FILE);
+    expect(schemaHasNote(vi.mocked(window.api.readMetadataBatch).mock.lastCall?.[1])).toBe(true);
+  });
+
+  it("(s6) A-R24-01: a dirty form keeps its unsaved edit when the schema re-read rejects", async () => {
+    seedFiles([PATH_A]);
+    vi.mocked(window.api.readMetadataBatch).mockRejectedValue(new Error("read down"));
+    render(createElement(MetadataPanel));
+    fireEvent.change(getInputByLabel("Category (UCS)"), { target: { value: TYPED_CATEGORY } });
+
+    await addNoteFieldAndDone();
+
+    const state = useMetadataStore.getState();
+    expect(state.toastMessage).toBe("Failed to read 2 files");
+    expect(getInputByLabel("Category (UCS)").value).toBe(TYPED_CATEGORY);
+    expect(state.isDirty).toBe(true);
+    expect(state.modifiedFields).toEqual(new Set(["category"]));
+    expect(useFileStore.getState().metadata).toEqual({ [PATH_A]: SEED_A, [PATH_B]: SEED_B });
+  });
+
+  it("(s7) A-R24-01: a title typed during a save survives a rejected schema re-read (D-30)", async () => {
+    seedFiles([PATH_A]);
+    vi.mocked(window.api.readMetadata).mockResolvedValue({ ...SEED_A, category: TYPED_CATEGORY });
+    vi.mocked(window.api.readMetadataBatch).mockRejectedValue(new Error("read down"));
+    const pending = deferSingleWrite();
+    render(createElement(MetadataPanel));
+
+    await typeCategoryAndSave(/Save Tags/);
+    fireEvent.change(getInputByLabel("Title"), { target: { value: "TypedDuring" } });
+    await addNoteFieldAndDone();
+    await settle(pending.release);
+
+    const state = useMetadataStore.getState();
+    expect(state.toastMessage).toBe("Tags saved");
+    expect(getInputByLabel("Title").value).toBe("TypedDuring");
+    expect(getInputByLabel("Category (UCS)").value).toBe(TYPED_CATEGORY);
+    expect(state.isDirty).toBe(true);
+    expect(useFileStore.getState().metadata[PATH_A]?.category).toBe(TYPED_CATEGORY);
+  });
+});
